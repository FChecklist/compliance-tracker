name: Reviewer Not Author Check

# Real, automated identity-mismatch guard (UMR-20260805-091629-d8e3, Owner directive,
# supporting the branch-protection hardening in UMR-20260805-034917-33a9 while
# UMR-20260805-091648-6793's real, temporary, bounded exception on
# required_approving_review_count is active). Fails CI (not a warning) if any APPROVED
# review on a PR was submitted by the same GitHub account (case-insensitive) as the
# PR's own author. Defense-in-depth: GitHub's own API already refuses a literal self-
# approval, so this exists to catch it anyway if something upstream goes wrong once a
# real distinct reviewer identity is wired in.
#
# See scripts/check-reviewer-not-author.mjs for the real gate logic and
# ai-os/REVIEWER_IDENTITY_PROVISIONING_GAP_2026-08-05.md for the real identity-
# provisioning gap this exists ahead of.

on:
  pull_request:
    branches: [main]
    types: [opened, synchronize, reopened]
  pull_request_review:
    types: [submitted]

permissions:
  contents: read
  pull-requests: read

jobs:
  reviewer-not-author-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
        with:
          fetch-depth: 0
      - uses: oven-sh/setup-bun@v2
      - run: bun install --frozen-lockfile

      - name: Run the gate's own real regression tests first
        run: bun test ./scripts/check-reviewer-not-author.test.ts

      - name: Evaluate reviewer != author identity for this PR
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: node scripts/check-reviewer-not-author.mjs "${{ github.event.pull_request.number }}"
